17    /// Moves a piece without restrictions. Also resets or increment the [`Game.halfmove_clock`].
18    pub(crate) fn move_piece(&mut self, original_position: &Position, new_position: &Position) {
19        if self.get_piece_at(original_position).unwrap().get_piece_type().eq(&Pawn) {
20            self.halfmove_clock = 0;
21        }
22        else if self.get_piece_at(new_position).is_some() {
23            self.halfmove_clock = 0;
24        }
25        else {
26            self.halfmove_clock += 1;
27        }
28
29        let piece = self.get_piece_at(original_position);
30
31        self.set_piece_at(new_position, piece);
32        self.set_piece_at(original_position, None);
33    }
34}

53/// Checks the board for a [`Pawn`] than can promote.

78/// Checks if the opponent is in [`Check`].
79/// See [`moves::king::king_illegal_moves`].
80///
81/// Returns true if the opponent [`King`] is in check.
82pub(crate) fn check_for_check(game: &Game) -> bool {
83    let mut king_piece_pos: String = "".to_string();
84
85    'outer: for x in 1..9 {
86        for y in 1..9 {
87            if let Some(piece) = game.get_piece_at(&Position::new(x, y)) && piece.get_piece_type().eq(&King) && piece.get_piece_color().eq(&game.get_turn().get_opponent_color()) {
88                king_piece_pos = Position::new(x,y).to_symbolic_representation();
89                break 'outer;
90            }
91
92        }
93    }
94
95    if moves::king::king_illegal_moves(&game, &game.get_turn()).contains(&king_piece_pos) {
96        true
97    }
98    else {
99        false
100    }
101}

115/// Checks if an en passant was performed. See [`handle_en_passant`]. This move only removes an additional piece, so we can still move the piece normally.
116/// Checks if a castling was performed. See [`handle_castling`]. If true, we don't want to move the piece normally, as castling provides a different movement pattern.
117///
118/// Else moves a piece normally. See [`Game::move_piece`].
119pub(crate) fn handle_move(game: &mut Game, from: &Position, to: &Position) {
120    handle_en_passant(game, &from, &to);
121    if !handle_castling(game, &from, &to) {
122        game.move_piece(&from, &to);
123    }
124
125}

127/// Checks if a move is an en passant, or allows for an en passant next turn.
128pub(crate) fn handle_en_passant(game: &mut Game, from: &Position, to: &Position) {
129    if let Some(piece) = game.get_piece_at(from) && piece.get_piece_type().eq(&Pawn) {
130
131        //En passant check
132        let color_offset: i8 = if game.get_turn().eq(&White) { -1 } else { 1 };
133        if let Some(position) = game.en_passant_position && to.eq(&position) {
134            game.set_piece_at(&position.add_y(color_offset), None);
135            game.en_passant_position = None;
136        }
137        else if (from.y - to.y).abs() == 2 { //Checks if a pawn has moved two steps
138            game.en_passant_position = Some(to.add_y(color_offset));
139        }
140    }
141    else {
142        game.en_passant_position = None;
143    }
144}

146/// Checks if castling still is allowed. See [`check_castling`]. Also handles a castling move.
147/// Returns true if the function handles castling, as it overrides the default movement behavior.
148///
149/// The function checks for the position of the rook. A castling can both happen from selecting the king or a rook, so we don't know which position contains the rook.
150/// See [`moves::rook::rook_castling`] and [`moves::king::king_castling`].
151pub(crate) fn handle_castling(game: &mut Game, from: &Position, to: &Position) -> bool {
152    let mut king_check = false;
153    let mut rook_position: Option<Position> = None;
154
155    if let Some(from_king) = game.get_piece_at(from) && from_king.get_piece_type().eq(&King) && from_king.get_piece_color().eq(&game.get_turn()) {
156        king_check = true;
157    }
158    else if let Some(to_king) = game.get_piece_at(to) && to_king.get_piece_type().eq(&King) && to_king.get_piece_color().eq(&game.get_turn()){
159        king_check = true;
160    }
161
162    if let Some(from_rook) = game.get_piece_at(from) && from_rook.get_piece_type().eq(&Rook) && from_rook.get_piece_color().eq(&game.get_turn()) {
163        rook_position = Some(*from);
164    }
165    else if let Some(to_rook) = game.get_piece_at(to) && to_rook.get_piece_type().eq(&Rook) && to_rook.get_piece_color().eq(&game.get_turn()) {
166        rook_position = Some(*to);
167    }
168
169    if let Some(position) = rook_position && king_check {
170        if position.eq(&Position::new(1, 1)) {
171            game.move_piece(&position, &Position::new(4, 1)); //Move rook
172            game.move_piece(&Position::new(5, 1), &Position::new(3, 1)); //Move kin
173
174            game.castling_king_white = false;
175            game.castling_queen_white = false;
176        }
177        else if position.eq(&Position::new(8, 1)) {
178            game.move_piece(&position, &Position::new(6, 1)); //Move rook
179            game.move_piece(&Position::new(5, 1), &Position::new(7, 1)); //Move king
180
181            game.castling_king_white = false;
182            game.castling_queen_white = false;
183        }
184
185        else if position.eq(&Position::new(1, 8)) {
186            game.move_piece(&position, &Position::new(4, 8)); //Move rook
187            game.move_piece(&Position::new(5, 8), &Position::new(3, 8)); //Move king
188
189            game.castling_king_black = false;
190            game.castling_queen_black = false;
191        }
192        else if position.eq(&Position::new(8, 8)) {
193            game.move_piece(&position, &Position::new(6, 8)); //Move rook
194            game.move_piece(&Position::new(5, 8), &Position::new(7, 8)); //Move king
195
196            game.castling_king_black = false;
197            game.castling_queen_black = false;
198        }
199
200        game.halfmove_clock -=1; //We move two pieces.
201        true
202    }
203    else {
204        check_castling(game, from);
205        false
206    }
207}

209/// Checks if castling still is allowed. Checks if a [`Rook`] has moved, and from what [`Position`], and if a king has moved and from what [`Position`].
210pub(crate) fn check_castling(game: &mut Game, from: &Position) {
211    if let Some(piece) = game.get_piece_at(from) {
212        if piece.get_piece_type().eq(&Rook) {
213            if from.eq(&Position::new(1, 1)) {
214                game.castling_queen_white = false;
215            } else if from.eq(&Position::new(8, 1)) {
216                game.castling_king_white = false;
217
218            }
219            if from.eq(&Position::new(1, 8)) {
220                game.castling_queen_black = false;
221
222            } else if from.eq(&Position::new(8, 8)) {
223                game.castling_king_black = false;
224            }
225        } else if piece.get_piece_type().eq(&King) {
226            if from.eq(&Position::new(5, 1)) {
227
228                game.castling_king_white = false;
229                game.castling_queen_white = false;
230            }
231            else if from.eq(&Position::new(5, 8)) {
232
233                game.castling_king_black = false;
234                game.castling_queen_black = false;
235            }
236        }
237    }
238}

240/// Checks various thing right after a turn.
241///
242/// Checks if any [`Pawn`] can promote. See [`check_for_promotion`].
243///
244/// Checks if the opponent is in [`Check`]. See [`check_for_check`].
245/// Checks if the opponent can do any legal moves. See [`has_opponent_possible_moves`].
246///
247/// If the opponent is in [`Check`] and does not have any legal moves. If true set [`game.state`] to [`Checkmate`].
248/// If the opponent isn't in [`Check`] and does not have any legal moves. If true set [`game.state`] to [`Stalemate`].
249///
250/// If the opponent can do any legal move, continues the game and changes turn. See [`Colour::get_opponent_color`].
251/// Also increment the [`Game::fullmove_clock`] if applicable.
252pub(crate) fn post_turn_check(game: &mut Game) {
253    if check_for_promotion(game).is_some() {
254        game.state = Promoting;
255        return;
256    }
257
258    game.state = if check_for_check(game) {Check} else {InProgress};
259    if !has_opponent_possible_moves(game) {
260        if game.state.eq(&Check) {
261            game.state = Checkmate;
262        }
263        else {
264            game.state = Stalemate;
265        }
266
267    }
268
269    if game.halfmove_clock >= 100 {
270        game.state = Stalemate;
271    }
272
273    if game.get_game_state().eq(&InProgress) || game.get_game_state().eq(&Check) {
274        if game.get_turn().eq(&Black) {
275            game.fullmove_counter += 1;
276        }
277
278        game.turn = game.turn.get_opponent_color();
279    }
280}

282/// Recursively goes through every possible move for every [`Piece`] on the board for the opposite player.
283/// Checks if the opponent can do any legal moves. Used to determinate [`Checkmate`] or [`Stalemate`].
284pub(crate) fn has_opponent_possible_moves(game: &mut Game) -> bool {
285    let mut game_test = game.clone();
286    game_test.turn = game.get_turn().get_opponent_color();
287
288    for x in 1..9 {
289        for y in 1..9 {
290            if let Some(piece) = game_test.get_piece_at(&Position::new(x, y)) && piece.get_piece_color().eq(&game_test.get_turn()) {
291                if !check_possible_moves(&Position::new(x, y), &game_test, &piece).is_empty() {
292                    return true
293                }
294            }
295        }
296    }
297    false
298}

300/// Checks if a [`Piece`] has any legal moves on a specific [`Position`].
301///
302/// Gets all moves a [`Piece`] can do, see [`Piece::get_possible_moves`] for per [`Type`] logic.
303///
304/// Checks if any moves puts the own [`King`] into check. See [`handle_move`] and [`check_for_check`].
305/// Moves that puts the own [`King`] into check are considered illegal and are not added to the final [`Vec<String>`] of allowed_moves.
306/// Returns a [`Vec<String>`] of all legal moves a [`Piece`] can do.
307pub(crate) fn check_possible_moves(position: &Position, game: &Game, piece: &Piece) -> Vec<String> {
308    let mut allowed_moves: Vec<String> = vec![];
309
310    for moves in piece.get_possible_moves(position, game) {
311
312        let mut game_test = game.clone();
313
314        handle_move(&mut game_test, position, &Position::from_symbolic_representation(&*moves));
315
316        game_test.turn = game.get_turn().get_opponent_color();
317        if !check_for_check(&game_test) {
318
319            if !allowed_moves.contains(&moves) {
320               allowed_moves.push(moves);
321            }
322        }
323    }
324
325    allowed_moves
326}
